# version 2.4
# Profile for python:3.11-slim containers

profile wol_forward_app_dev flags=(attach_disconnected, mediate_deleted) {
  
  # 1. Standaard Docker abstracties importeren
  include <abstractions/base>
  include <abstractions/nameservice>

  # 2. Netwerkmogelijkheden (aanpassen naar behoefte)
  network inet stream,
  network inet6 stream,

  # 3. Kernel mogelijkheden (Capabilities ontzeggen)
  capability chown,
  capability setuid,
  capability setgid,
  deny capability sys_admin,
  deny capability sys_chroot,
  deny capability ptrace,

  # 4. Systeembestanden (Alleen lezen voor Python runtime runtime-omgeving)
  /bin/** rmix,
  /usr/bin/** rmix,
  /usr/local/bin/python3* rmix,
  /lib/** r,
  /usr/lib/** r,
  /usr/local/lib/python3.11/** r,

  # 5. Essentiële systeempaden voor Python
  /etc/ld.so.cache r,
  /etc/localtime r,
  /etc/timezone r,
  /dev/urandom r,
  /dev/random r,
  /dev/null rw,
  /dev/zero rw,

  # 6. Applicatie-specifieke paden (Wijzig '/app' naar jouw WORKDIR)
  /app/ r,
  /app/** r,
  
  # Geef schrijfrechten waar strikt noodzakelijk (bijv. tijdelijke bestanden)
  /tmp/ rw,
  /tmp/** rwk,

  # 7. Expliciete blokkades voor gevoelige locaties
  deny /proc/** w,
  deny /sys/** w,
  deny /etc/passwd* w,
  deny /etc/shadow* w,
}
